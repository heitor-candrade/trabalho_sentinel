\section{Métricas principais}\label{ap:metricas}

\begin{center}
\small
\begin{tabularx}{\linewidth}{@{}>{\raggedright}p{6cm}>{\raggedright}p{1.7cm}>{\raggedright\arraybackslash}X@{}}
\toprule
\textbf{Métrica (rótulos)} & \textbf{Tipo} & \textbf{Significado}\\ \midrule
\cmd{sentinel_http_requests_total} (\cmd{node,route,code}) & contador & requisições por rota e código\\
\cmd{sentinel_app_request_duration_seconds} (\cmd{node,le}) & histograma & latência do usuário (\cmd{/}, \cmd{/erro}) medida \emph{no nó}\\
\cmd{sentinel_ready}, \cmd{sentinel_inflight_requests} & medidor & prontidão; requisições em andamento (saturação)\\
\cmd{sentinel_uptime_seconds} & medidor & tempo desde o início do processo (detecta reinício)\\
\cmd{watchdog_target_down} (\cmd{target}) & medidor & visão do watchdog: 1 = DOWN\\
\cmd{watchdog_consecutive_failures} & medidor & falhas consecutivas atuais\\
\cmd{watchdog_last_probe_timestamp_seconds} & medidor & instante da última sondagem; idade = \cmd{time() - ...}\\
\cmd{watchdog_probe_failures_total} (\cmd{target,reason}) & contador & falhas por motivo (\cmd{timeout}, \cmd{http_503}, \cmd{no_route}, \cmd{refused}\ldots)\\
\cmd{watchdog_state_changes_total} (\cmd{target,to}) & contador & transições UP/DOWN (detecta \emph{flapping})\\
\cmd{up} (\cmd{job,instance}) & medidor & do Prometheus: 1 se a coleta funcionou\\ \bottomrule
\end{tabularx}
\end{center}

\section{PromQL mínimo}

\begin{center}
\small
\begin{tabularx}{\linewidth}{@{}>{\raggedright}p{8.2cm}X@{}}
\toprule
\textbf{Consulta} & \textbf{O que responde}\\ \midrule
\cmd{watchdog_target_down == 1} & quais nós o watchdog considera DOWN\\
\cmd{rate(sentinel_http_requests_total{route="root"}[1m])} & requisições por segundo por nó e código\\
\cmd{histogram_quantile(0.95, sum by (le,node) (rate(sentinel_app_request_duration_seconds_bucket[1m])))} & p95 da latência por nó\\
\cmd{time() - watchdog_last_probe_timestamp_seconds} & idade (s) da última sondagem\\
\cmd{absent(up{job="watchdog"})} & retorna 1 \emph{se a série não existe}: útil quando a ausência é o problema\\ \bottomrule
\end{tabularx}
\end{center}

\section{Problemas comuns}

\begin{center}
\small
\begin{tabularx}{\linewidth}{@{}>{\raggedright}p{5cm}>{\raggedright\arraybackslash}X@{}}
\toprule
\textbf{Sintoma} & \textbf{Causa provável e solução}\\ \midrule
\cmd{port is already allocated} & porta 8080/9090/8085 ocupada: mude \cmd{HOST_PORT} ou \cmd{PAINEL_PORT} no \cmd{.env}\\ \midrule
Painel em branco ou ``sem dados'' & \cmd{docker compose logs painel}; o painel depende do watchdog (\cmd{http://localhost:9100} só na rede interna); \cmd{docker compose restart painel}\\ \midrule
Quero recomeçar do zero & \cmd{scripts/remover-tudo.sh --simular} (mostra o que seria removido), depois sem \cmd{--simular}\\ \midrule
\cmd{Pool overlaps with other one} & sub-rede em conflito: mude \cmd{NET_PREFIX} no \cmd{.env} e \cmd{scripts/setup.sh --limpo}\\ \midrule
\cmd{permission denied ... docker.sock} & usuário fora do grupo \cmd{docker}: \cmd{sudo usermod -aG docker $USER} e novo login\\ \midrule
\cmd{chaos.sh}: ``modulo ausente'' & \cmd{sudo apt install linux-modules-extra-$(uname -r)}; \cmd{sudo modprobe sch_netem sch_prio cls_u32}\\ \midrule
Nginx ou alerta não mudou após editar & \cmd{scripts/validar.sh --recarregar}; \cmd{docker compose logs nginx}; confira \cmd{http://localhost:9090/rules}\\ \midrule
Experimento ``sujo'' (falha anterior ativa) & \cmd{scripts/chaos.sh limpar todos}; \cmd{scripts/admin.sh atraso rust-a 0}; \cmd{scripts/admin.sh ready rust-a 1}\\ \midrule
Tudo estranho & \cmd{scripts/evidence.sh antes-de-resetar} e depois \cmd{scripts/setup.sh --limpo}\\ \bottomrule
\end{tabularx}
\end{center}
